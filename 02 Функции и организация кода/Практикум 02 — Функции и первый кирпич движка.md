> namespace {
>
> // Выбросить остаток строки после неудачного чтения

>     std::cin.clear();
>     std::cin.ignore(std::numeric_limits<std::streamsize>::max(), '\n');

>     if (c >= 'A' && c <= 'Z') {

>     int value = 0;
>     while (!(std::cin >> value) || value < min || value > max) {

>         }
>         discardLine();
>         std::print("Нужно число от {} до {}: ", min, max);
>     }
>     return value;
> }

> char readChoice(const std::string& allowed) {
>     char c = '\0';
>     while (std::cin >> c) {
>         c = toLower(c);
>         if (allowed.find(c) != std::string::npos) {
>             return c;
>         }
>         discardLine();

> [!solution]- Решение: Арена 2.0
> `practice/m02_functions/arena/arena.cpp`:
> ```cpp
> #include <console/input.h>
>
> #include <algorithm>
> #include <print>
> #include <random>
> #include <string>
> #include <vector>
>
> struct Monster {
>     std::string name;
>     int health = 0;
>     int maxHealth = 0;
>     int minDamage = 0;
>     int maxDamage = 0;
> };
>
> struct Player {
>     int health = 100;
>     int maxHealth = 100;
>     int potions = 3;
>     int totalDealt = 0;
> };
>
> // ---------- Вывод ----------
>
> void printBar(const std::string& label, int value, int maxValue) {
>     int filled = std::max(value, 0) * 20 / maxValue;
>     std::println("{:<6} [{:-<20}] {:3}/{}", label, std::string(filled, '#'), value, maxValue);
> }
>
> void printStatus(int turn, const Player& player, const Monster& monster) {
>     std::println("\n--- Ход {} ---  (зелий: {})", turn, player.potions);
>     printBar("Ты", player.health, player.maxHealth);
>     printBar(monster.name, monster.health, monster.maxHealth);
> }
>
> // ---------- Логика ----------
>
> Monster pickMonster(std::mt19937& rng) {
>     const std::vector<Monster> bestiary = {
>         {"Imp", 60, 60, 4, 9},
>         {"Pinky", 90, 90, 6, 12},
>         {"Baron", 140, 140, 8, 16},
>     };
>     std::uniform_int_distribution<int> pick(0, static_cast<int>(bestiary.size()) - 1);
>     return bestiary[pick(rng)];
> }
>
> void attack(Player& player, Monster& monster, std::mt19937& rng) {
>     std::uniform_int_distribution<int> damageDist(8, 15);
>     std::bernoulli_distribution critical(0.15);
>
>     int damage = damageDist(rng);
>     if (critical(rng)) {
>         damage *= 2;
>         std::println("\x1b[33mКРИТИЧЕСКИЙ УДАР!\x1b[0m");
>     }
>     monster.health -= damage;
>     player.totalDealt += damage;
>     std::println("Ты наносишь {} урона.", damage);
> }
>
> void drinkPotion(Player& player) {
>     if (player.potions == 0) {
>         std::println("Зелья кончились! Ход потерян.");
>         return;
>     }
>     --player.potions;
>     int healed = std::min(30, player.maxHealth - player.health);
>     player.health += healed;
>     std::println("Ты выпиваешь зелье: +{} здоровья.", healed);
> }
>
> // Возвращает true, если игрок в этот ход защищается
> bool playerTurn(Player& player, Monster& monster, std::mt19937& rng) {
>     std::print("1) атака  2) защита  3) зелье > ");
>     switch (console::readInt(1, 3)) {
>         case 1:
>             attack(player, monster, rng);
>             return false;
>         case 2:
>             std::println("Ты поднимаешь щит.");
>             return true;
>         default:
>             drinkPotion(player);
>             return false;
>     }
> }
>
> void monsterTurn(const Monster& monster, Player& player, bool defending, std::mt19937& rng) {
>     std::uniform_int_distribution<int> damageDist(monster.minDamage, monster.maxDamage);
>     std::bernoulli_distribution rage(0.5);
>
>     int damage = damageDist(rng);
>     bool wounded = monster.health * 100 < monster.maxHealth * 30;
>     if (wounded && rage(rng)) {
>         damage = damage * 3 / 2;
>         std::println("\x1b[31m{} впадает в ярость!\x1b[0m", monster.name);
>     }
>     if (defending) {
>         damage /= 2;
>     }
>     player.health -= damage;
>     std::println("{} бьёт тебя на {} урона.", monster.name, damage);
> }
>
> // ---------- Игра ----------
>
> int main() {
>     std::mt19937 rng(std::random_device{}());
>
>     Player player;
>     Monster monster = pickMonster(rng);
>     std::println("На арену выходит {}! Здоровье: {}", monster.name, monster.health);
>
>     int turn = 0;
>     while (player.health > 0 && monster.health > 0) {
>         ++turn;
>         printStatus(turn, player, monster);
>         bool defending = playerTurn(player, monster, rng);
>         if (monster.health > 0) {
>             monsterTurn(monster, player, defending, rng);
>         }
>     }
>
>     if (player.health > 0) {
>         std::println("\n\x1b[32mПобеда!\x1b[0m {} повержен за {} ходов. Нанесено урона: {}.",
>                      monster.name, turn, player.totalDealt);
>     } else {
>         std::println("\n\x1b[31mПоражение.\x1b[0m {} оказался сильнее. Ходов продержался: {}.",
>                      monster.name, turn);
>     }
> }
> ```
>
> В `m02_functions/CMakeLists.txt`:
> ```cmake
> add_executable(m02_arena arena/arena.cpp)
> target_link_libraries(m02_arena PRIVATE console)
> ```
>
> **Почему генератор по ссылке.** Если написать `void attack(Player& player, Monster& monster, std::mt19937 rng)`, функция получит **копию** генератора со всем его состоянием. Она выдаст «случайное» число из копии, а копия умрёт в конце функции. Генератор в `main` при этом **не сдвинется**, и следующий вызов скопирует то же самое состояние и выдаст **то же самое** число. Урон будет одинаковым весь бой. Это реальный баг, который встречается в коде новичков.
>
> **Что стало лучше.** `main` читается как описание игры: выбрать монстра, пока оба живы — показать состояние, ход игрока, ход монстра. Каждую функцию можно понять, не читая остальные. А `readInt` из библиотеки пригодится в рогалике и Сапёре.

## Задание 2. Математика Ember и первые тесты ⭐⭐

> [!task] Условие
> Расширь математическую библиотеку Ember из [[02.6 CMake — библиотеки и структура проекта#Рождение Ember|урока 02.6]].
>
> **Скалярные функции:**
> - `clamp(value, lo, hi)` — ограничить значение диапазоном;
> - `lerp(a, b, t)` — **линейная интерполяция**: `a + (b - a) * t`. При `t = 0` даёт `a`, при `t = 1` — `b`, при `t = 0.5` — середину;
> - `nearlyEqual(a, b, epsilon = 1e-5f)` — сравнение дробных чисел с допуском ([[01.2 Переменные и фундаментальные типы#Почему 0.1 + 0.2 != 0.3|урок 01.2]]).
>
> **Для `Vec2`** (к уже существующим `add`, `sub`, `scale`, `length`):
> - `dot(a, b)` — **скалярное произведение**: `a.x * b.x + a.y * b.y`;
> - `lengthSquared(v)` — квадрат длины (без корня);
> - `distance(a, b)` — расстояние между точками;
> - `normalize(v)` — вектор длины 1 того же направления; для нулевого вектора вернуть `(0, 0)`;
> - `lerp(a, b, t)` для векторов;
> - `nearlyEqual(a, b, epsilon)` для векторов.
>
> **Новый тип `Vec2i`** — целочисленный вектор для клеток сетки — с функциями `add` и `equal`.
>
> **Тесты.** Создай программу `ember_tests` (файл `practice/ember/tests/math_tests.cpp`, цель — в `ember/CMakeLists.txt`). В ней функция `check(bool condition, const std::string& what)`, которая считает проверки и печатает сообщение о каждой проваленной. Программа проверяет все функции библиотеки и возвращает из `main` **`0`, если всё прошло, и `1`, если хоть что-то провалилось**.
>
> Пометь все функции, которые только вычисляют результат, `[[nodiscard]]`.

> [!hint]- Подсказка: что проверять
> - Удобные «круглые» векторы: `(3, 4)` — длина 5, нормализованный `(0.6, 0.8)`.
> - Перпендикулярные векторы `(1, 0)` и `(0, 1)` — скалярное произведение 0.
> - Граничные случаи: `clamp` ниже, внутри и выше диапазона; `lerp` при `t = 0` и `t = 1`; `normalize` нулевого вектора.
> - Векторы сравнивай через `nearlyEqual`, а не `==`: после `sqrt` и деления точное совпадение не гарантировано.

> [!solution]- Решение
> `practice/ember/include/ember/math.h`:

>

>

> `practice/ember/src/math.cpp`:
> ```cpp
> #include <ember/math.h>
>
> #include <cmath>
>
> namespace ember {
>
> // ---------- Скалярные функции ----------
>
> float clamp(float value, float lo, float hi) {
>     if (value < lo) {
>         return lo;
>     }
>     if (value > hi) {
>         return hi;
>     }
>     return value;
> }
>
> float lerp(float a, float b, float t) {
>     return a + (b - a) * t;
> }
>
> bool nearlyEqual(float a, float b, float epsilon) {
>     return std::abs(a - b) <= epsilon;
> }
>
> // ---------- Vec2 ----------
>
> Vec2 add(Vec2 a, Vec2 b) {
>     return {a.x + b.x, a.y + b.y};
> }
>
> Vec2 sub(Vec2 a, Vec2 b) {
>     return {a.x - b.x, a.y - b.y};
> }
>
> Vec2 scale(Vec2 v, float factor) {
>     return {v.x * factor, v.y * factor};
> }
>
> float dot(Vec2 a, Vec2 b) {
>     return a.x * b.x + a.y * b.y;
> }
>
> float lengthSquared(Vec2 v) {
>     return dot(v, v);
> }
>
> float length(Vec2 v) {
>     return std::sqrt(lengthSquared(v));
> }
>
> float distance(Vec2 a, Vec2 b) {
>     return length(sub(b, a));
> }
>
> Vec2 normalize(Vec2 v) {
>     float len = length(v);
>     if (len < 1e-8f) {
>         return {0.0f, 0.0f};    // у нулевого вектора нет направления
>     }
>     float inv = 1.0f / len;     // одно деление вместо двух
>     return {v.x * inv, v.y * inv};
> }
>
> Vec2 lerp(Vec2 a, Vec2 b, float t) {
>     return {lerp(a.x, b.x, t), lerp(a.y, b.y, t)};
> }
>
> bool nearlyEqual(Vec2 a, Vec2 b, float epsilon) {
>     return nearlyEqual(a.x, b.x, epsilon) && nearlyEqual(a.y, b.y, epsilon);
> }
>
> // ---------- Vec2i ----------
>
> Vec2i add(Vec2i a, Vec2i b) {
>     return {a.x + b.x, a.y + b.y};
> }
>
> bool equal(Vec2i a, Vec2i b) {
>     return a.x == b.x && a.y == b.y;
> }
>
> }  // namespace ember

> [!hint]- Подсказка
> Итеративная заливка:
> 1. положить стартовую клетку в вектор;
> 2. пока вектор не пуст: снять последнюю клетку (`back()` и `pop_back()`); если её можно залить — залить и положить в вектор четырёх соседей.
>
> Это та же рекурсия, только «стек вызовов» теперь твой собственный вектор в куче, а куча — это гигабайты, а не мегабайт.

> [!solution]- Решение
> `practice/m02_functions/flood.cpp`:
> ```cpp
> #include <ember/math.h>
>
> #include <print>
> #include <random>
> #include <string>
> #include <vector>
>
> namespace {
>
> using Grid = std::vector<std::string>;
>
> // Случайная карта: рамка из стен, внутри стена с вероятностью wallChance
> Grid generateMap(int width, int height, float wallChance, std::mt19937& rng) {
>     std::bernoulli_distribution isWall(wallChance);
>     Grid map(height, std::string(width, '#'));
>     for (int y = 1; y < height - 1; ++y) {
>         for (int x = 1; x < width - 1; ++x) {
>             map[y][x] = isWall(rng) ? '#' : '.';
>         }
>     }
>     return map;
> }
>
> ember::Vec2i randomFloor(const Grid& map, std::mt19937& rng) {
>     std::uniform_int_distribution<int> randomX(0, static_cast<int>(map[0].size()) - 1);
>     std::uniform_int_distribution<int> randomY(0, static_cast<int>(map.size()) - 1);
>     while (true) {
>         ember::Vec2i p{randomX(rng), randomY(rng)};
>         if (map[p.y][p.x] == '.') {
>             return p;
>         }
>     }
> }
>
> bool canFill(const Grid& map, int x, int y) {
>     return y >= 0 && y < static_cast<int>(map.size()) &&
>            x >= 0 && x < static_cast<int>(map[y].size()) &&
>            map[y][x] == '.';
> }
>
> // Рекурсивная заливка: просто и наглядно, но глубина рекурсии растёт с размером области
> int floodFillRecursive(Grid& map, int x, int y) {
>     if (!canFill(map, x, y)) {
>         return 0;
>     }
>     map[y][x] = '~';
>     return 1 + floodFillRecursive(map, x + 1, y) + floodFillRecursive(map, x - 1, y) +
>                floodFillRecursive(map, x, y + 1) + floodFillRecursive(map, x, y - 1);
> }
>
> // Та же заливка с явным стеком в куче: глубина не ограничена стеком потока
> int floodFillIterative(Grid& map, int startX, int startY) {
>     std::vector<ember::Vec2i> stack;
>     stack.push_back({startX, startY});
>     int filled = 0;
>     while (!stack.empty()) {
>         ember::Vec2i p = stack.back();
>         stack.pop_back();
>         if (!canFill(map, p.x, p.y)) {
>             continue;
>         }
>         map[p.y][p.x] = '~';
>         ++filled;
>         stack.push_back({p.x + 1, p.y});
>         stack.push_back({p.x - 1, p.y});
>         stack.push_back({p.x, p.y + 1});
>         stack.push_back({p.x, p.y - 1});
>     }
>     return filled;
> }
>

>             if (x == start.x && y == start.y) {

>                 frame += "\x1b[32;1m>\x1b[0m";
>             } else if (map[y][x] == '~') {
>                 frame += "\x1b[34m~\x1b[0m";
>             } else if (map[y][x] == '#') {
>                 frame += "\x1b[90m#\x1b[0m";
>             } else {
>                 frame += map[y][x];
>             }
>         }
>         frame += '\n';
>     }
>     std::print("{}", frame);
> }
>
> }  // namespace
>
> int main() {

>     int reachable = floodFillRecursive(map, start.x, start.y);

>     std::println("Достижимо клеток: {} из {}", reachable, floorCells);
>     std::println("{}", map[exit.y][exit.x] == '~' ? "Выход достижим!" : "Выход НЕдостижим: нужна другая карта.");

>     // ---------- Часть 2: огромная пустая карта ----------

>     // Двумерное поле хранится в одномерных массивах: индекс клетки = y * width + x
>     std::vector<char> mines;     // 1 — мина (не vector<bool>, см. практикум 01)
>     std::vector<int> adjacent;   // сколько мин в 8 соседних клетках
>     std::vector<int> state;      // CellHidden / CellOpen / CellFlagged
> };
>
> [[nodiscard]] Board createBoard(int width, int height, int mineCount);
> [[nodiscard]] bool isInside(const Board& board, int x, int y);
> [[nodiscard]] int cellIndex(const Board& board, int x, int y);
>
> // Расставить мины, не трогая клетку (safeX, safeY) и её соседей: первый ход всегда безопасен
> void placeMines(Board& board, int safeX, int safeY, std::mt19937& rng);
>
> // Открыть клетку. Пустые клетки (0 мин рядом) открывают соседей рекурсивно.
> // Возвращает false, если под клеткой мина.
> bool openCell(Board& board, int x, int y);
>
> void toggleFlag(Board& board, int x, int y);
> [[nodiscard]] bool isWon(const Board& board);
> void drawBoard(const Board& board, bool revealMines);
> ```
> `practice/m02_functions/minesweeper/board.cpp`:
> ```cpp
> #include "board.h"
>
> #include <array>
> #include <cstdlib>
> #include <format>
> #include <print>
> #include <string>
>
> Board createBoard(int width, int height, int mineCount) {
>     Board board;
>     board.width = width;
>     board.height = height;
>     board.mineCount = mineCount;
>     board.mines.assign(width * height, 0);
>     board.adjacent.assign(width * height, 0);
>     board.state.assign(width * height, CellHidden);
>     return board;
> }
>
> bool isInside(const Board& board, int x, int y) {
>     return x >= 0 && y >= 0 && x < board.width && y < board.height;
> }
>
> int cellIndex(const Board& board, int x, int y) {
>     return y * board.width + x;
> }
>
> void placeMines(Board& board, int safeX, int safeY, std::mt19937& rng) {
>     std::uniform_int_distribution<int> randomCell(0, board.width * board.height - 1);
>     int placed = 0;
>     while (placed < board.mineCount) {
>         int i = randomCell(rng);
>         int x = i % board.width;
>         int y = i / board.width;
>         bool nearSafe = std::abs(x - safeX) <= 1 && std::abs(y - safeY) <= 1;
>         if (board.mines[i] != 0 || nearSafe) {
>             continue;
>         }
>         board.mines[i] = 1;
>         ++placed;
>     }
>
>     // Посчитать мины вокруг каждой клетки
>     for (int y = 0; y < board.height; ++y) {
>         for (int x = 0; x < board.width; ++x) {
>             int count = 0;
>             for (int dy = -1; dy <= 1; ++dy) {
>                 for (int dx = -1; dx <= 1; ++dx) {
>                     if (isInside(board, x + dx, y + dy) && board.mines[cellIndex(board, x + dx, y + dy)] != 0) {
>                         ++count;
>                     }
>                 }
>             }
>             board.adjacent[cellIndex(board, x, y)] = count;
>         }
>     }
>     board.minesPlaced = true;
> }
>
> bool openCell(Board& board, int x, int y) {
>     if (!isInside(board, x, y)) {
>         return true;
>     }
>     int i = cellIndex(board, x, y);
>     if (board.state[i] != CellHidden) {
>         return true;    // уже открыта или под флагом
>     }
>     board.state[i] = CellOpen;
>     if (board.mines[i] != 0) {
>         return false;
>     }
>     if (board.adjacent[i] == 0) {
>         for (int dy = -1; dy <= 1; ++dy) {
>             for (int dx = -1; dx <= 1; ++dx) {
>                 if (dx != 0 || dy != 0) {
>                     openCell(board, x + dx, y + dy);
>                 }
>             }
>         }
>     }
>     return true;
> }
>
> void toggleFlag(Board& board, int x, int y) {
>     int i = cellIndex(board, x, y);
>     if (board.state[i] == CellHidden) {
>         board.state[i] = CellFlagged;
>     } else if (board.state[i] == CellFlagged) {
>         board.state[i] = CellHidden;
>     }
> }
>
> bool isWon(const Board& board) {
>     for (int i = 0; i < board.width * board.height; ++i) {
>         if (board.mines[i] == 0 && board.state[i] != CellOpen) {
>             return false;
>         }
>     }
>     return true;
> }
>
> void drawBoard(const Board& board, bool revealMines) {
>     // Цвета цифр, как в классическом Сапёре: 1 синий, 2 зелёный, 3 красный...
>     const std::array<std::string, 9> digitColors = {"0", "34", "32", "31", "35", "33", "36", "37", "90"};
>
>     std::string frame = "\x1b[2J\x1b[H   ";
>     for (int x = 0; x < board.width; ++x) {
>         frame += std::format("{:2}", x);
>     }
>     frame += '\n';
>
>     for (int y = 0; y < board.height; ++y) {
>         frame += std::format("{:2} ", y);
>         for (int x = 0; x < board.width; ++x) {
>             int i = cellIndex(board, x, y);
>             if (board.mines[i] != 0 && (revealMines || board.state[i] == CellOpen)) {
>                 frame += " \x1b[31;1m*\x1b[0m";
>             } else if (board.state[i] == CellFlagged) {
>                 frame += " \x1b[91mF\x1b[0m";
>             } else if (board.state[i] == CellHidden) {
>                 frame += " \x1b[90m#\x1b[0m";
>             } else if (board.adjacent[i] == 0) {
>                 frame += " .";
>             } else {
>                 frame += std::format(" \x1b[{}m{}\x1b[0m", digitColors[board.adjacent[i]], board.adjacent[i]);
>             }
>         }
>         frame += '\n';
>     }
>     std::print("{}", frame);
> }
> ```
> `practice/m02_functions/minesweeper/main.cpp`:
> ```cpp
> #include "board.h"
>
> #include <console/input.h>
>
> #include <print>
> #include <random>
>
> int main() {
>     std::mt19937 rng(std::random_device{}());
>
>     std::print("Сапёр. Поле: 1) 9x9, 10 мин  2) 16x16, 40 мин > ");
>     Board board = console::readInt(1, 2) == 1 ? createBoard(9, 9, 10) : createBoard(16, 16, 40);
>
>     while (true) {
>         drawBoard(board, false);
>         std::print("Ход: o x y — открыть, f x y — флаг, q — выход > ");
>         char command = console::readChoice("ofq");
>         if (command == 'q' || command == '\0') {
>             return 0;
>         }
>         int x = console::readInt(0, board.width - 1);
>         int y = console::readInt(0, board.height - 1);
>
>         if (command == 'f') {
>             toggleFlag(board, x, y);
>             continue;
>         }
>         if (!board.minesPlaced) {
>             placeMines(board, x, y, rng);   // мины ставим после первого хода
>         }
>         if (!openCell(board, x, y)) {
>             drawBoard(board, true);
>             std::println("\x1b[31mБУМ!\x1b[0m Ты наступил на мину.");
>             return 0;
>         }
>         if (isWon(board)) {
>             drawBoard(board, true);
>             std::println("\x1b[32mПобеда!\x1b[0m Все безопасные клетки открыты.");
>             return 0;
>         }
>     }
> }
> ```
>
> В `m02_functions/CMakeLists.txt`:
> ```cmake
> add_executable(m02_minesweeper minesweeper/main.cpp minesweeper/board.cpp)
> target_link_libraries(m02_minesweeper PRIVATE console)
> ```
>
> **Почему одномерные массивы.** Двумерное поле можно хранить как `std::vector<std::vector<int>>`, но тогда каждая строка — отдельный буфер в куче, разбросанный по памяти. Один вектор `width * height` лежит в памяти **одним куском**, а формула `y * width + x` — это ровно то, что делает компилятор с `std::array<std::array<...>>`. Так хранят изображения, карты высот, сетки навигации и framebuffer нашего будущего рендерера. Привыкай к этой формуле: в модуле 07 она будет в каждой второй строке.
>
> **Формат ввода.** `readChoice` и `readInt` читают по одному «слову», поэтому `o 4 4` можно ввести одной строкой, а можно по частям.

## Задание 6. Детектив — линковщик ⭐⭐⭐

> [!task] Условие
> Программа считает статистику раунда. Создай папку `m02_functions/detective/`, а в ней четыре файла **ровно в таком виде** (ошибки в них специально):
>
> `stats.h`:
> ```cpp
> struct Stats {
>     int kills = 0;
>     int deaths = 0;
> };
>
> int totalKills = 0;
> ```
>
> `report.h`:
> ```cpp
> #pragma once
> #include "stats.h"
>
> void printReport(const Stats& stats);
>
> float killDeathRatio(const Stats& stats) {
>     if (stats.deaths == 0) {
>         return static_cast<float>(stats.kills);
>     }
>     return static_cast<float>(stats.kills) / static_cast<float>(stats.deaths);
> }
> ```
>
> `report.cpp`:
> ```cpp
> #include "report.h"
>
> #include <print>
>
> void printReport(Stats& stats) {
>     totalKills += stats.kills;
>     std::println("Убийств: {}, смертей: {}, K/D: {:.2f}", stats.kills, stats.deaths, killDeathRatio(stats));
> }
> ```
>
> `main.cpp`:
> ```cpp
> #include "stats.h"
> #include "report.h"
>
> #include <print>
>
> int main() {
>     Stats round{25, 4};
>     printReport(round);
>     std::println("Всего убийств за сессию: {}", totalKills);
> }
> ```
>
> И цель в `m02_functions/CMakeLists.txt`:
> ```cmake
> add_executable(m02_detective detective/main.cpp)
> ```
>
> Собирай, читай ошибки и исправляй их **по одной**, каждый раз записывая: код ошибки, этап (компиляция или линковка) и причину. В этом проекте **пять** ошибок. Ожидаемый результат после всех исправлений:
> ```
> Убийств: 25, смертей: 4, K/D: 6.25
> Всего убийств за сессию: 25
> ```

> [!hint]- Подсказка
> Ошибки проявляются волнами: пока не исправлены ошибки компиляции, линковщик вообще не запускается, и его ошибок ты не увидишь. Подсказки по номерам: заголовок без защиты; файл, забытый в CMake; две сущности, определённые в заголовке; объявление и определение, которые выглядят одинаково, но не совпадают.

> [!solution]- Решение
> **Волна 1, компиляция:**
> ```
> stats.h(1): error C2011: Stats: переопределение типа "struct"
> stats.h(6): error C2374: totalKills: переопределение; множественная инициализация
> ```
> `main.cpp` подключает `stats.h` дважды: напрямую и через `report.h`. **Ошибка 1:** в `stats.h` нет `#pragma once`. Добавь его первой строкой. (Остальные ошибки компиляции в этой волне, вроде `C2027: использование неопределенного типа`, — следствия первой. Всегда исправляй **первую** ошибку в списке и пересобирай.)
>
> **Волна 2, линковка:**
> ```
> main.obj : error LNK2019: ссылка на неразрешенный внешний символ "void __cdecl printReport(struct Stats const &)" ...
> ```
> **Ошибка 2:** `report.cpp` не добавлен в цель. Исправь CMake:
> ```cmake
> add_executable(m02_detective detective/main.cpp detective/report.cpp)
> ```
>
> **Волна 3, линковка:**
> ```
> report.obj : error LNK2005: "int totalKills" (?totalKills@@3HA) уже определен в main.obj
> report.obj : error LNK2005: "float __cdecl killDeathRatio(struct Stats const &)" (...) уже определен в main.obj
> main.obj : error LNK2019: ссылка на неразрешенный внешний символ "void __cdecl printReport(struct Stats const &)" ...
> ```
> - **Ошибка 3:** переменная `totalKills` определена в заголовке, и теперь, когда в сборке два `.cpp`, определений два. Исправление: `inline int totalKills = 0;` (или `extern int totalKills;` в заголовке плюс `int totalKills = 0;` в `report.cpp`).
> - **Ошибка 4:** функция `killDeathRatio` определена в заголовке без `inline`. Исправление: `inline float killDeathRatio(...)` или перенос тела в `report.cpp`.
> - **Ошибка 5:** самая коварная. `LNK2019` про `printReport` никуда не делся, хотя `report.cpp` теперь в сборке. Сравни декорированное имя в ошибке (`Stats const &`) с определением в `report.cpp`: там `Stats&`, без `const`. Это **другая перегрузка**. Исправление: `void printReport(const Stats& stats)`.
>
> **Мораль.**
> - Исправляй ошибки по одной, начиная с первой.
> - `#pragma once` — в каждом заголовке, без исключений.
> - Ошибка линковщика про функцию, которая «точно есть», — почти всегда несовпадение сигнатур. Читай подпись в сообщении внимательно, вплоть до `const` и `&`.
